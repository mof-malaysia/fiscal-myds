{/* Story links work here and composed into the React Storybook, where ids
    carry the ref's prefix (astro_core-button--primary) and the page lives in
    the parent Storybook's frame. */}

export const storyUrl = (id) => {
  const ref = new URLSearchParams(window.location.search).get("refId");
  try {
    return new URL(
      `./?path=/story/${ref ? `${ref}_` : ""}${id}`,
      window.top.location.href,
    ).href;
  } catch {
    // Cross-origin parent (local dev composition): open this Storybook itself
    return new URL(`./?path=/story/${id}`, window.location.href).href;
  }
};

export const StoryLink = ({ id, className, children }) => (
  <a
    href={storyUrl(id)}
    target="_top"
    className={className ?? "text-txt-primary underline"}
  >
    {children}
  </a>
);
